// Draws the key glyphs as fonts. node make.mjs. y up, 1000 to the em, Special Gothic's cap height 710.
import opentype from 'opentype.js'; import { writeFileSync } from 'node:fs';
const CAP = 710;
// ── strokes ──
const sub = (a, b) => [a[0] - b[0], a[1] - b[1]], len = (v) => Math.hypot(v[0], v[1]), unit = (v) => { const l = len(v) || 1; return [v[0] / l, v[1] / l]; }, nor = (v) => [-v[1], v[0]];
const side = (pts, d, closed) => pts.map((p, i) => { const prev = closed ? pts[(i - 1 + pts.length) % pts.length] : pts[i - 1]; const next = closed ? pts[(i + 1) % pts.length] : pts[i + 1]; const n1 = prev ? nor(unit(sub(p, prev))) : null, n2 = next ? nor(unit(sub(next, p))) : null; if (!n1 || !n2) { const n = n1 ?? n2; return [p[0] + n[0] * d, p[1] + n[1] * d]; } const m = unit([n1[0] + n2[0], n1[1] + n2[1]]); const k = d / Math.max(0.3, m[0] * n1[0] + m[1] * n1[1]); return [p[0] + m[0] * k, p[1] + m[1] * k]; });
const area = (c) => c.reduce((s, p, i) => { const q = c[(i + 1) % c.length]; return s + p[0] * q[1] - q[0] * p[1]; }, 0) / 2;
const cw = (c) => (area(c) < 0 ? c : [...c].reverse()), ccw = (c) => (area(c) > 0 ? c : [...c].reverse());
const disc = (p, r) => cw(Array.from({ length: 28 }, (_, i) => [p[0] + Math.cos((i / 28) * 2 * Math.PI) * r, p[1] + Math.sin((i / 28) * 2 * Math.PI) * r]));
const open = (pts, s) => cw([...side(pts, s / 2, false), ...side(pts, -s / 2, false).reverse()]);
const line = (pts, s, round) => round ? [open(pts, s), ...pts.filter((p, i) => i === 0 || i === pts.length - 1 || sharp(pts, i)).map((p) => disc(p, s / 2))] : [open(pts, s)];
const sharp = (pts, i) => { const a = unit(sub(pts[i], pts[i - 1])), b = unit(sub(pts[i + 1], pts[i])); return a[0] * b[0] + a[1] * b[1] < 0.95; };
const roundOpen = (pts, s) => { const out = []; for (let i = 1; i < pts.length; i++) out.push(open([pts[i - 1], pts[i]], s)); pts.forEach((p) => out.push(disc(p, s / 2))); return out; };
const arc = (c, r, a0, a1, n = 30) => Array.from({ length: n + 1 }, (_, i) => { const t = ((a0 + ((a1 - a0) * i) / n) * Math.PI) / 180; return [c[0] + Math.cos(t) * r, c[1] + Math.sin(t) * r]; });
// an outline: the outer shape, and its hole s in
const shell = (outer, s) => { const o = ccw(outer); return [cw(outer), ccw(side(o, s, true))]; };
const shellRound = (outer, s) => { const mid = side(ccw(outer), s / 2, true); return roundOpen([...mid, mid[0]], s); };
const move = (cs, dx, dy) => cs.map((c) => c.map(([x, y]) => [x + dx, y + dy]));
// ── the glyphs ── each returns [contours, width]
const G = {
  shift: (s, R) => { const h = R ? CAP + 30 : CAP, w = R ? 770 : 760, sh = h * 0.5, c = w / 2, t = w * 0.255; const o = [[c, h], [w, h - sh], [c + t, h - sh], [c + t, 0], [c - t, 0], [c - t, h - sh], [0, h - sh]]; return [move((R ? shellRound : shell)(o, s), 0, (CAP - h) / 2), w]; },
  cmd: (s, R) => { const H = (R ? 730 : CAP) / 2, a = R ? 92 : 88, r = (H - s / 2 - a) / 2; const out = []; for (const [X, Y] of [[1, 1], [-1, 1], [-1, -1], [1, -1]]) { const start = X > 0 ? (Y > 0 ? 180 : 90) : Y > 0 ? 270 : 0; const pts = arc([X * (a + r), Y * (a + r)], r, start, start - 270 * (1), 60); out.push(open(pts, s)); } for (const q of [a, -a]) { out.push(open([[-(a + r) - 1, q], [a + r + 1, q]], s)); out.push(open([[q, -(a + r) - 1], [q, a + r + 1]], s)); } return [move(out, H, CAP / 2), H * 2]; },
  opt: (s, R) => { const T = CAP - s / 2 - (R ? 20 : 30), B = s / 2 + (R ? 20 : 30), w = 740, e = R ? s / 2 : 0; const L = [[e, T], [w * 0.3, T], [w * 0.64, B], [w - e, B]], bar = [[w * 0.6, T], [w - e, T]]; return [R ? [...roundOpen(L, s), ...roundOpen(bar, s)] : [...line(L, s), ...line(bar, s)], w]; },
  ctrl: (s, R) => { const w = 560, y0 = 330, y1 = 600; const p = R ? [[s / 2, y0], [w / 2, y1], [w - s / 2, y0]] : [[s * 0.36, y0], [w / 2, y1], [w - s * 0.36, y0]]; return [R ? roundOpen(p, s) : line(p, s), w]; },
  ret: (s, R) => { const w = 720, y = 250, T = 610, r = 130, e = R ? s / 2 : s * 0.7; const p = [[w - s / 2, T], [w - s / 2, y + r], ...arc([w - s / 2 - r, y + r], r, 0, -90, 14).slice(1), [e, y]]; const hd = [[e + 190, y + 195], [e, y], [e + 190, y - 195]]; return [R ? [...roundOpen(p, s), ...roundOpen(hd, s)] : [open(p, s), ...line(hd, s)], w]; },
  del: (s, R) => { const w = 860, h = 630, y = (CAP - h) / 2; const o = [[0, y + h / 2], [270, y + h], [w, y + h], [w, y], [270, y]]; const x = 535, k = 118; const X = [[[x - k, CAP / 2 - k], [x + k, CAP / 2 + k]], [[x - k, CAP / 2 + k], [x + k, CAP / 2 - k]]]; return [[...(R ? shellRound : shell)(o, s), ...X.flatMap((l) => (R ? roundOpen(l, s) : line(l, s)))], w]; },
  tab: (s, R) => { const w = 760, y = CAP / 2, e = w - s * 1.6; const p = [[R ? s / 2 : 0, y], [e - (R ? 0 : s * 0.5), y]], hd = [[e - 190, y + 195], [e, y], [e - 190, y - 195]], bar = [[w - s / 2, y + 220], [w - s / 2, y - 220]]; return [R ? [...roundOpen(p, s), ...roundOpen(hd, s), ...roundOpen(bar, s)] : [...line(p, s), ...line(hd, s), ...line(bar, s)], w]; },
  left: (s, R) => { const w = 690, y = CAP / 2, e = R ? s / 2 : s * 0.7; const p = [[w - (R ? s / 2 : 0), y], [e, y]], hd = [[e + 200, y + 205], [e, y], [e + 200, y - 205]]; return [R ? [...roundOpen(p, s), ...roundOpen(hd, s)] : [...line(p, s), ...line(hd, s)], w]; },
  right: (s, R) => { const [c, w] = G.left(s, R); return [c.map((q) => cw(q.map(([x, y]) => [w - x, y]))), w]; },
  up: (s, R) => { const [c, w] = G.left(s, R); return [c.map((q) => cw(q.map(([x, y]) => [y - CAP / 2 + 205 + s * 0.5, CAP / 2 + w / 2 - x]))), 410 + s]; },
  down: (s, R) => { const [c, w] = G.up(s, R); return [c.map((q) => cw(q.map(([x, y]) => [x, CAP - y]))), w]; },
};
const CODE = { shift: 0x21e7, cmd: 0x2318, opt: 0x2325, ctrl: 0x2303, ret: 0x21a9, del: 0x232b, tab: 0x21e5, left: 0x2190, right: 0x2192, up: 0x2191, down: 0x2193 };
const SETS = { A: { R: false, own: ['shift', 'cmd', 'opt', 'ctrl', 'ret', 'del', 'tab'], w: { 400: 78, 500: 106, 600: 120 } }, B: { R: true, own: Object.keys(CODE), w: { 400: 76, 500: 96, 600: 108 } } };
const svg = {};
for (const [name, set] of Object.entries(SETS)) for (const [w, s] of Object.entries(set.w)) {
  const glyphs = [new opentype.Glyph({ name: '.notdef', unicode: 0, advanceWidth: 500, path: new opentype.Path() })];
  for (const g of set.own) { const k = g === 'cmd' || g === 'del' || g === 'shift' ? 0.92 : 1; const [cs, width] = G[g](s * k, set.R); const p = new opentype.Path(); for (const c of cs) { c.forEach(([x, y], i) => (i ? p.lineTo(x + 70, y) : p.moveTo(x + 70, y))); p.close(); } glyphs.push(new opentype.Glyph({ name: g, unicode: CODE[g], advanceWidth: Math.round(width + 140), path: p })); svg[`${name}${w}${g}`] = [p.toPathData(1), width + 140]; }
  const f = new opentype.Font({ familyName: `Interlude Keys ${name}`, styleName: 'Regular', unitsPerEm: 1000, ascender: 1000, descender: -250, glyphs }); const buf = Buffer.from(f.toArrayBuffer()); writeFileSync(`keys-${name}-${w}.otf`, buf); console.log(name, w, buf.length, 'bytes');
}
writeFileSync('svg.json', JSON.stringify(svg));
